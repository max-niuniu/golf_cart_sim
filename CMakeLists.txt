cmake_minimum_required(VERSION 3.8)
project(golf_cart_sim)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(gazebo REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ackermann_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
add_library(golf_ackermann SHARED src/ackermann_plugin.cpp)
target_include_directories(golf_ackermann PRIVATE ${GAZEBO_INCLUDE_DIRS})
target_link_libraries(golf_ackermann ${GAZEBO_LIBRARIES})
ament_target_dependencies(golf_ackermann gazebo_ros rclcpp ackermann_msgs nav_msgs sensor_msgs tf2_msgs)
target_compile_options(golf_ackermann PRIVATE -Wall -Wextra)
find_program(XACRO_EXECUTABLE xacro REQUIRED)
set(GOLF_URDF ${CMAKE_CURRENT_BINARY_DIR}/golf_cart.urdf)
add_custom_command(OUTPUT ${GOLF_URDF}
  COMMAND ${XACRO_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/urdf/golf_cart.urdf.xacro
    config_file:=${CMAKE_CURRENT_SOURCE_DIR}/config/vehicle.yaml -o ${GOLF_URDF}
  DEPENDS urdf/golf_cart.urdf.xacro config/vehicle.yaml VERBATIM)
add_custom_target(golf_description ALL DEPENDS ${GOLF_URDF})
install(TARGETS golf_ackermann LIBRARY DESTINATION lib)
install(DIRECTORY config urdf launch worlds meshes DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE PATTERN "*.pyc" EXCLUDE)
install(FILES ${GOLF_URDF} DESTINATION share/${PROJECT_NAME}/urdf)
install(FILES README.md README.zh-CN.md DESTINATION share/${PROJECT_NAME})
install(DIRECTORY docs DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/drive.py DESTINATION lib/${PROJECT_NAME} RENAME drive)
install(PROGRAMS scripts/control_ui.py DESTINATION lib/${PROJECT_NAME} RENAME control_ui)
ament_environment_hooks(${CMAKE_CURRENT_SOURCE_DIR}/env-hooks/golf_cart_sim.dsv)
ament_package()
